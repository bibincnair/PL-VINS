cmake_minimum_required(VERSION 3.22)
project(feature_tracker VERSION 0.1.0 LANGUAGES CXX)
include(${CMAKE_CURRENT_SOURCE_DIR}/../cmake/Dependencies.cmake)
find_package(ament_cmake REQUIRED)
find_package(plvins_common REQUIRED)
find_package(camera_model REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(OpenCV 5.0.0 EXACT REQUIRED COMPONENTS core imgproc video features calib geometry imgcodecs highgui)
add_library(plvins_line_descriptor STATIC src/line_descriptor/src/binary_descriptor_custom.cpp src/line_descriptor/src/binary_descriptor_matcher.cpp src/line_descriptor/src/draw_custom.cpp src/line_descriptor/src/LSDDetector_custom.cpp)
target_include_directories(plvins_line_descriptor PUBLIC src/line_descriptor/include)
target_link_libraries(plvins_line_descriptor PUBLIC ${OpenCV_LIBS})
add_executable(feature_tracker src/feature_tracker_node.cpp src/parameters.cpp src/feature_tracker.cpp)
add_executable(LineFeature_tracker src/linefeature_tracker_node.cpp src/parameters.cpp src/linefeature_tracker.cpp)
foreach(target feature_tracker LineFeature_tracker)
  ament_target_dependencies(${target} plvins_common camera_model rclcpp sensor_msgs geometry_msgs std_msgs cv_bridge)
  target_link_libraries(${target} Eigen3::Eigen ${OpenCV_LIBS})
endforeach()
target_link_libraries(LineFeature_tracker plvins_line_descriptor)
install(TARGETS feature_tracker LineFeature_tracker DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(line_descriptor_test test/line_descriptor_test.cpp)
  target_link_libraries(line_descriptor_test plvins_line_descriptor)
endif()
ament_package()
