cmake_minimum_required(VERSION 3.22)
project(plvins_line_descriptor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 5 REQUIRED COMPONENTS core imgproc features highgui)
add_library(linedesc SHARED src/binary_descriptor_custom.cpp src/binary_descriptor_matcher.cpp src/draw_custom.cpp src/LSDDetector_custom.cpp)
target_include_directories(linedesc PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(linedesc PUBLIC ${OpenCV_LIBS})
install(TARGETS linedesc LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
